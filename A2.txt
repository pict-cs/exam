#include <bits/stdc++.h>
#include <omp.h>

using namespace std;
using namespace chrono;

// ---------------- PRINT ARRAY ----------------
void printArray(vector<int>& arr)
{
    for (int x : arr)
    {
        cout << x << " ";
    }

    cout << endl;
}

// ---------------- GENERATE RANDOM ARRAY ----------------
void generateArray(vector<int>& arr, int size)
{
    arr.resize(size);

    for (int i = 0; i < size; i++)
    {
        arr[i] = rand() % 1000;
    }
}

// ---------------- SEQUENTIAL BUBBLE SORT ----------------
void sequentialBubbleSort(vector<int> arr)
{
    auto start = high_resolution_clock::now();

    int n = arr.size();

    for (int i = 0; i < n - 1; i++)
    {
        for (int j = 0; j < n - i - 1; j++)
        {
            if (arr[j] > arr[j + 1])
            {
                swap(arr[j], arr[j + 1]);
            }
        }
    }

    auto end = high_resolution_clock::now();

    cout << "\nSequential Bubble Sort Time: "
         << duration_cast<microseconds>(end - start).count()
         << " microseconds\n";

    // Uncomment if needed
    // printArray(arr);
}

// ---------------- PARALLEL BUBBLE SORT ----------------
// Using Odd-Even Transposition Sort
void parallelBubbleSort(vector<int> arr)
{
    auto start = high_resolution_clock::now();

    int n = arr.size();

    for (int phase = 0; phase < n; phase++)
    {
        // Even Phase
        if (phase % 2 == 0)
        {
            #pragma omp parallel for
            for (int i = 0; i < n - 1; i += 2)
            {
                if (arr[i] > arr[i + 1])
                {
                    swap(arr[i], arr[i + 1]);
                }
            }
        }

        // Odd Phase
        else
        {
            #pragma omp parallel for
            for (int i = 1; i < n - 1; i += 2)
            {
                if (arr[i] > arr[i + 1])
                {
                    swap(arr[i], arr[i + 1]);
                }
            }
        }
    }

    auto end = high_resolution_clock::now();

    cout << "Parallel Bubble Sort Time: "
         << duration_cast<microseconds>(end - start).count()
         << " microseconds\n";

    // Uncomment if needed
    // printArray(arr);
}

// ---------------- MERGE FUNCTION ----------------
void merge(vector<int>& arr, int low, int mid, int high)
{
    vector<int> temp(high - low + 1);

    int i = low;
    int j = mid + 1;
    int k = 0;

    while (i <= mid && j <= high)
    {
        if (arr[i] <= arr[j])
        {
            temp[k++] = arr[i++];
        }
        else
        {
            temp[k++] = arr[j++];
        }
    }

    while (i <= mid)
    {
        temp[k++] = arr[i++];
    }

    while (j <= high)
    {
        temp[k++] = arr[j++];
    }

    for (int i = 0; i < temp.size(); i++)
    {
        arr[low + i] = temp[i];
    }
}

// ---------------- SEQUENTIAL MERGE SORT ----------------
void sequentialMergeSort(vector<int>& arr, int low, int high)
{
    if (low < high)
    {
        int mid = (low + high) / 2;

        sequentialMergeSort(arr, low, mid);
        sequentialMergeSort(arr, mid + 1, high);

        merge(arr, low, mid, high);
    }
}

// ---------------- PARALLEL MERGE SORT ----------------
const int THRESHOLD = 1000;

void parallelMergeSort(vector<int>& arr, int low, int high)
{
    if (low < high)
    {
        // Small array -> sequential
        if (high - low < THRESHOLD)
        {
            sequentialMergeSort(arr, low, high);
            return;
        }

        int mid = (low + high) / 2;

        #pragma omp task shared(arr)
        parallelMergeSort(arr, low, mid);

        #pragma omp task shared(arr)
        parallelMergeSort(arr, mid + 1, high);

        #pragma omp taskwait

        merge(arr, low, mid, high);
    }
}

// ---------------- MAIN ----------------
int main()
{
    srand(time(0));

    int size;

    cout << "Enter size of array: ";
    cin >> size;

    vector<int> original;

    generateArray(original, size);

    vector<int> arr1 = original;
    vector<int> arr2 = original;
    vector<int> arr3 = original;
    vector<int> arr4 = original;

    omp_set_num_threads(4);

    // Small array -> print
    if (size <= 20)
    {
        cout << "\nOriginal Array:\n";
        printArray(original);
    }

    // ---------------- Sequential Bubble Sort ----------------
    sequentialBubbleSort(arr1);

    // ---------------- Parallel Bubble Sort ----------------
    parallelBubbleSort(arr2);

    // ---------------- Sequential Merge Sort ----------------
    auto start = high_resolution_clock::now();

    sequentialMergeSort(arr3, 0, size - 1);

    auto end = high_resolution_clock::now();

    cout << "Sequential Merge Sort Time: "
         << duration_cast<microseconds>(end - start).count()
         << " microseconds\n";

    // ---------------- Parallel Merge Sort ----------------
    start = high_resolution_clock::now();

    #pragma omp parallel
    {
        #pragma omp single
        parallelMergeSort(arr4, 0, size - 1);
    }

    end = high_resolution_clock::now();

    cout << "Parallel Merge Sort Time: "
         << duration_cast<microseconds>(end - start).count()
         << " microseconds\n";

    // Optional sorted array printing
    if (size <= 20)
    {
        cout << "\nSorted Array:\n";
        printArray(arr4);
    }

    return 0;
}
