#include <iostream>
#include <vector>
#include <queue>
#include <stack>
#include <omp.h>
#include <chrono>
#include <cstdlib>
#include <ctime>

using namespace std;
using namespace chrono;

class Graph
{
    int V;
    vector<vector<int>> adj;

public:

    // Constructor
    Graph(int vertices)
    {
        V = vertices;
        adj.resize(V);
    }

    // Add Edge
    void addEdge(int u, int v)
    {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    // Print Graph
    void printGraph()
    {
        cout << "\nAdjacency List:\n";

        for (int i = 0; i < V; i++)
        {
            cout << i << " -> ";

            for (int neighbor : adj[i])
            {
                cout << neighbor << " ";
            }

            cout << endl;
        }
    }

    // Random Graph Generation
    void generateRandomGraph(int edges)
    {
        srand(time(0));

        for (int i = 0; i < edges; i++)
        {
            int u = rand() % V;
            int v = rand() % V;

            if (u != v)
            {
                addEdge(u, v);
            }
        }
    }

    // ---------------- SEQUENTIAL BFS ----------------
    void sequentialBFS(int start, bool printTraversal)
    {
        vector<bool> visited(V, false);

        queue<int> q;

        visited[start] = true;
        q.push(start);

        if (printTraversal)
            cout << "\nSequential BFS Traversal: ";

        while (!q.empty())
        {
            int current = q.front();
            q.pop();

            if (printTraversal)
                cout << current << " ";

            for (int neighbor : adj[current])
            {
                if (!visited[neighbor])
                {
                    visited[neighbor] = true;
                    q.push(neighbor);
                }
            }
        }

        if (printTraversal)
            cout << endl;
    }

    // ---------------- PARALLEL BFS ----------------
    void parallelBFS(int start, bool printTraversal)
    {
        vector<int> visited(V, 0);

        vector<int> current;
        current.push_back(start);

        visited[start] = 1;

        if (printTraversal)
            cout << "\nParallel BFS Traversal: ";

        while (!current.empty())
        {
            vector<int> next;

            #pragma omp parallel for
            for (int i = 0; i < current.size(); i++)
            {
                int node = current[i];

                if (printTraversal)
                {
                    #pragma omp critical
                    cout << node << " ";
                }

                for (int neighbor : adj[node])
                {
                    if (!visited[neighbor])
                    {
                        #pragma omp critical
                        {
                            if (!visited[neighbor])
                            {
                                visited[neighbor] = 1;
                                next.push_back(neighbor);
                            }
                        }
                    }
                }
            }

            current = next;
        }

        if (printTraversal)
            cout << endl;
    }

    // ---------------- SEQUENTIAL DFS ----------------
    void sequentialDFS(int start, bool printTraversal)
    {
        vector<bool> visited(V, false);

        stack<int> s;

        s.push(start);

        if (printTraversal)
            cout << "\nSequential DFS Traversal: ";

        while (!s.empty())
        {
            int current = s.top();
            s.pop();

            if (!visited[current])
            {
                visited[current] = true;

                if (printTraversal)
                    cout << current << " ";

                for (int neighbor : adj[current])
                {
                    if (!visited[neighbor])
                    {
                        s.push(neighbor);
                    }
                }
            }
        }

        if (printTraversal)
            cout << endl;
    }

    // ---------------- PARALLEL DFS ----------------
    void parallelDFS(int start, bool printTraversal)
    {
        vector<bool> visited(V, false);

        stack<int> s;

        s.push(start);

        if (printTraversal)
            cout << "\nParallel DFS Traversal: ";

        while (!s.empty())
        {
            int current;

            #pragma omp critical
            {
                current = s.top();
                s.pop();
            }

            if (!visited[current])
            {
                visited[current] = true;

                if (printTraversal)
                    cout << current << " ";

                #pragma omp parallel for
                for (int i = 0; i < adj[current].size(); i++)
                {
                    int neighbor = adj[current][i];

                    if (!visited[neighbor])
                    {
                        #pragma omp critical
                        {
                            s.push(neighbor);
                        }
                    }
                }
            }
        }

        if (printTraversal)
            cout << endl;
    }
};

int main()
{
    int choice;

    while (true)
    {
        cout << "\n1. Manual Graph Input\n";
        cout << "2. Random Graph Generation\n";
        cout << "3. Exit\n";

        cout << "\nEnter choice: ";
        cin >> choice;

        if (choice == 3)
        {
            cout << "\nExiting Program...\n";
            break;
        }

        int V, E;

        Graph *g;

        if (choice == 1)
        {
            cout << "\nEnter number of vertices: ";
            cin >> V;

            g = new Graph(V);

            cout << "Enter number of edges: ";
            cin >> E;

            cout << "Enter edges (u v):\n";

            for (int i = 0; i < E; i++)
            {
                int u, v;
                cin >> u >> v;

                g->addEdge(u, v);
            }

            int start;

            cout << "Enter starting vertex: ";
            cin >> start;

            g->printGraph();

            // Sequential BFS
            auto start_time = high_resolution_clock::now();

            g->sequentialBFS(start, true);

            auto end_time = high_resolution_clock::now();

            cout << "Sequential BFS Time: "
                 << duration_cast<microseconds>(end_time - start_time).count()
                 << " microseconds\n";

            // Parallel BFS
            start_time = high_resolution_clock::now();

            g->parallelBFS(start, true);

            end_time = high_resolution_clock::now();

            cout << "Parallel BFS Time: "
                 << duration_cast<microseconds>(end_time - start_time).count()
                 << " microseconds\n";

            // Sequential DFS
            start_time = high_resolution_clock::now();

            g->sequentialDFS(start, true);

            end_time = high_resolution_clock::now();

            cout << "Sequential DFS Time: "
                 << duration_cast<microseconds>(end_time - start_time).count()
                 << " microseconds\n";

            // Parallel DFS
            start_time = high_resolution_clock::now();

            g->parallelDFS(start, true);

            end_time = high_resolution_clock::now();

            cout << "Parallel DFS Time: "
                 << duration_cast<microseconds>(end_time - start_time).count()
                 << " microseconds\n";
        }

        else if (choice == 2)
        {
            cout << "\nEnter number of vertices: ";
            cin >> V;

            cout << "Enter number of edges: ";
            cin >> E;

            g = new Graph(V);

            g->generateRandomGraph(E);

            int start = 0;

            cout << "\nLarge random graph generated successfully.\n";
            cout << "Traversal output hidden for large graph.\n";

            // Sequential BFS
            auto start_time = high_resolution_clock::now();

            g->sequentialBFS(start, false);

            auto end_time = high_resolution_clock::now();

            cout << "\nSequential BFS Time: "
                 << duration_cast<microseconds>(end_time - start_time).count()
                 << " microseconds\n";

            // Parallel BFS
            start_time = high_resolution_clock::now();

            g->parallelBFS(start, false);

            end_time = high_resolution_clock::now();

            cout << "Parallel BFS Time: "
                 << duration_cast<microseconds>(end_time - start_time).count()
                 << " microseconds\n";

            // Sequential DFS
            start_time = high_resolution_clock::now();

            g->sequentialDFS(start, false);

            end_time = high_resolution_clock::now();

            cout << "Sequential DFS Time: "
                 << duration_cast<microseconds>(end_time - start_time).count()
                 << " microseconds\n";

            // Parallel DFS
            start_time = high_resolution_clock::now();

            g->parallelDFS(start, false);

            end_time = high_resolution_clock::now();

            cout << "Parallel DFS Time: "
                 << duration_cast<microseconds>(end_time - start_time).count()
                 << " microseconds\n";
        }

        else
        {
            cout << "\nInvalid Choice\n";
        }
    }

    return 0;
}
