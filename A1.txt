#include <bits/stdc++.h>
#include <omp.h>

using namespace std;
using namespace chrono;

class Graph
{
    int V;
    vector<vector<int>> adj;

public:
    Graph(int vertices)
    {
        V = vertices;
        adj.resize(V);
    }

    // Add edge
    void addEdge(int u, int v)
    {
        adj[u].push_back(v);
        adj[v].push_back(u); // Undirected Graph
    }

    // Print graph
    void printGraph()
    {
        cout << "\nAdjacency List:\n";

        for (int i = 0; i < V; i++)
        {
            cout << i << " -> ";

            for (int neighbor : adj[i])
            {
                cout << neighbor << " ";
            }

            cout << endl;
        }
    }

    // ---------------- SEQUENTIAL BFS ----------------
    void sequentialBFS(int start)
    {
        vector<bool> visited(V, false);

        queue<int> q;

        visited[start] = true;
        q.push(start);

        cout << "\nSequential BFS Traversal: ";

        while (!q.empty())
        {
            int current = q.front();
            q.pop();

            cout << current << " ";

            for (int neighbor : adj[current])
            {
                if (!visited[neighbor])
                {
                    visited[neighbor] = true;
                    q.push(neighbor);
                }
            }
        }

        cout << endl;
    }

    // ---------------- PARALLEL BFS ----------------
    void parallelBFS(int start)
    {
        vector<bool> visited(V, false);

        queue<int> q;

        visited[start] = true;
        q.push(start);

        cout << "\nParallel BFS Traversal:";

        while (!q.empty())
        {
            int current;

            #pragma omp critical
            {
                if (!q.empty())
                {
                    current = q.front();
                    q.pop();

                    cout << current << " ";
                }
            }

            #pragma omp parallel for
            for (int i = 0; i < adj[current].size(); i++)
            {
                int neighbor = adj[current][i];

                bool shouldPush = false;

                #pragma omp critical
                {
                    if (!visited[neighbor])
                    {
                        visited[neighbor] = true;
                        shouldPush = true;
                    }
                }

                if (shouldPush)
                {
                    #pragma omp critical
                    {
                        q.push(neighbor);
                    }
                }
            }
        }
    }

    // ---------------- SEQUENTIAL DFS ----------------
    void sequentialDFS(int start)
    {
        vector<bool> visited(V, false);

        stack<int> s;

        s.push(start);

        cout << "\nSequential DFS Traversal: ";

        while (!s.empty())
        {
            int current = s.top();
            s.pop();

            if (!visited[current])
            {
                visited[current] = true;

                cout << current << " ";

                for (int neighbor : adj[current])
                {
                    if (!visited[neighbor])
                    {
                        s.push(neighbor);
                    }
                }
            }
        }

        cout << endl;
    }

    // ---------------- PARALLEL DFS ----------------
    void parallelDFS(int start)
    {
        vector<bool> visited(V, false);

        stack<int> s;

        s.push(start);

        cout << "\nParallel DFS Traversal:";

        while (!s.empty())
        {
            int current;

            #pragma omp critical
            {
                if (!s.empty())
                {
                    current = s.top();
                    s.pop();
                }
            }

            bool processNode = false;

            #pragma omp critical
            {
                if (!visited[current])
                {
                    visited[current] = true;
                    processNode = true;

                    cout << current << " ";
                }
            }

            if (processNode)
            {
                #pragma omp parallel for
                for (int i = 0; i < adj[current].size(); i++)
                {
                    int neighbor = adj[current][i];

                    bool shouldPush = false;

                    #pragma omp critical
                    {
                        if (!visited[neighbor])
                        {
                            shouldPush = true;
                        }
                    }

                    if (shouldPush)
                    {
                        #pragma omp critical
                        {
                            s.push(neighbor);
                        }
                    }
                }
            }
        }
    }
};

int main()
{
    int choice;

    while (true)
    {
        cout << "\n1. Run Graph Traversal\n";
        cout << "2. Exit\n";

        cout << "\nEnter choice: ";
        cin >> choice;

        if (choice == 2)
        {
            cout << "\nExiting Program\n";
            break;
        }

        else if (choice == 1)
        {
            int V, E;

            cout << "Enter number of vertices: ";
            cin >> V;

            Graph g(V);

            cout << "Enter number of edges: ";
            cin >> E;

            cout << "Enter edges (u v):\n";

            for (int i = 0; i < E; i++)
            {
                int u, v;
                cin >> u >> v;

                g.addEdge(u, v);
            }

            int start;

            cout << "Enter starting vertex: ";
            cin >> start;

            g.printGraph();

            // ---------------- Sequential BFS ----------------
            auto start_time = high_resolution_clock::now();

            g.sequentialBFS(start);

            auto end_time = high_resolution_clock::now();

            auto duration =
                duration_cast<microseconds>(end_time - start_time);

            cout << "Sequential BFS Time: "
                 << duration.count()
                 << " microseconds\n";

            // ---------------- Parallel BFS ----------------
            start_time = high_resolution_clock::now();

            g.parallelBFS(start);

            end_time = high_resolution_clock::now();

            duration =
                duration_cast<microseconds>(end_time - start_time);

            cout << "\nParallel BFS Time: "
                 << duration.count()
                 << " microseconds\n";

            // ---------------- Sequential DFS ----------------
            start_time = high_resolution_clock::now();

            g.sequentialDFS(start);

            end_time = high_resolution_clock::now();

            duration =
                duration_cast<microseconds>(end_time - start_time);

            cout << "Sequential DFS Time: "
                 << duration.count()
                 << " microseconds\n";

            // ---------------- Parallel DFS ----------------
            start_time = high_resolution_clock::now();

            g.parallelDFS(start);

            end_time = high_resolution_clock::now();

            duration =
                duration_cast<microseconds>(end_time - start_time);

            cout << "\nParallel DFS Time: "
                 << duration.count()
                 << " microseconds\n";
        }

        else
        {
            cout << "\nInvalid Choice\n";
        }
    }

    return 0;
}
